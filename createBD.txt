
-- ========================================
-- Création des tables
-- ========================================

CREATE TABLE typeProduit (
    id integer PRIMARY KEY auto_increment,
    libelle VARCHAR(20)
);

CREATE TABLE typePate (
    id integer PRIMARY KEY auto_increment,
    libelle VARCHAR(20) NOT NULL
);

CREATE TABLE typePizza (
    id integer PRIMARY KEY auto_increment,
    libelle VARCHAR(20)
);

CREATE TABLE produit (
    id integer PRIMARY KEY auto_increment,
    libelle VARCHAR(40),
    description VARCHAR(150),
    quantite INTEGER,
    prix DOUBLE PRECISION,
    prixLarge DOUBLE PRECISION,
    centilitre VARCHAR(10),
    urlImage VARCHAR(100),
    idTypePizza INTEGER,
    idTypeProduit INTEGER,
    CONSTRAINT fk_produit_typepizza FOREIGN KEY (idTypePizza) REFERENCES typePizza(id),
    CONSTRAINT fk_produit_typeproduit FOREIGN KEY (idTypeProduit) REFERENCES typeProduit(id)
);

CREATE TABLE categorieIngredient (
    id integer PRIMARY KEY auto_increment,
    libelle VARCHAR(20)
);

CREATE TABLE ingredient (
    id integer PRIMARY KEY auto_increment,
    libelle VARCHAR(40) NOT NULL,
    prixMedium DOUBLE PRECISION NOT NULL,
    prixLarge DOUBLE PRECISION NOT NULL,
    forPizza BOOLEAN NOT NULL,
    forPate BOOLEAN NOT NULL,
    forSalade BOOLEAN NOT NULL,
    idCategorieIngredient INTEGER NOT NULL,
    CONSTRAINT fk_ingredient_categorie FOREIGN KEY (idCategorieIngredient) REFERENCES categorieIngredient(id)
);

CREATE TABLE adresse (
    id integer PRIMARY KEY auto_increment,
    codePostal INTEGER NOT NULL,
    nomRue VARCHAR(25) NOT NULL,
    numeroRue VARCHAR(10) NOT NULL,
    localite VARCHAR(25) NOT NULL
);

CREATE TABLE client (
    id integer PRIMARY KEY auto_increment,
    nom VARCHAR(25) NOT NULL,
    prenom VARCHAR(25),
    numero VARCHAR(15) NOT NULL,
    idAdresse INTEGER NOT NULL,
    CONSTRAINT fk_client_adresse FOREIGN KEY (idAdresse) REFERENCES adresse(id)
);

CREATE TABLE restaurant (
    id integer PRIMARY KEY auto_increment,
    nom VARCHAR(20) NOT NULL,
    idAdresse INTEGER NOT NULL,
    CONSTRAINT fk_restaurant_adresse FOREIGN KEY (idAdresse) REFERENCES adresse(id)
);

CREATE TABLE typePayement (
    id integer PRIMARY KEY auto_increment,
    libelle VARCHAR(20) NOT NULL
);

CREATE TABLE statut (
    id integer PRIMARY KEY auto_increment,
    libelle VARCHAR(15)
);

CREATE TABLE commande (
    id integer PRIMARY KEY auto_increment,
    estLivraison BOOLEAN NOT NULL,
    dateHeure VARCHAR(20) NOT NULL,
    estPayer BOOLEAN NOT NULL,
    prixTotal DOUBLE PRECISION NOT NULL,
    commetaire VARCHAR(255),
    idrestaurant INTEGER NOT NULL,
    idstatut INTEGER NOT NULL,
    idClient INTEGER,
    idTypePayement INTEGER,
    CONSTRAINT fk_commande_restaurant FOREIGN KEY (idrestaurant) REFERENCES restaurant(id),
    CONSTRAINT fk_commande_statut FOREIGN KEY (idstatut) REFERENCES statut(id),
    CONSTRAINT fk_commande_client FOREIGN KEY (idClient) REFERENCES client(id),
    CONSTRAINT fk_commande_typepayement FOREIGN KEY (idTypePayement) REFERENCES typePayement(id)
);

CREATE TABLE ligneCommande (
    id integer PRIMARY KEY auto_increment,
    quantiteLigne INT NOT NULL,
    prixLigne DOUBLE PRECISION NOT NULL,
    estGratiner BOOLEAN,
    tailleMedium BOOLEAN,
    typePate VARCHAR(20),
    idProduit INTEGER,
    idCommande INTEGER,
    CONSTRAINT fk_ligne_produit FOREIGN KEY (idProduit) REFERENCES produit(id),
    CONSTRAINT fk_ligne_commande FOREIGN KEY (idCommande) REFERENCES commande(id)
);

CREATE TABLE ligneIngrediant (
    idLigne INTEGER NOT NULL,
    idIngredient INTEGER NOT NULL,
    ingredientEnPlus BOOLEAN NOT NULL,
    CONSTRAINT fk_ligneingrediant_ligne FOREIGN KEY (idLigne) REFERENCES ligneCommande(id),
    CONSTRAINT fk_ligneingrediant_ingredient FOREIGN KEY (idIngredient) REFERENCES ingredient(id)
);

CREATE TABLE utilisateur (
    id integer PRIMARY KEY  auto_increment,
    mdp VARCHAR(10),
    idResraurent INTEGER,
    CONSTRAINT fk_utilisateur_restaurant FOREIGN KEY (idResraurent) REFERENCES restaurant(id)
);
